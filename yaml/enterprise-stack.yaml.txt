apiVersion: v1
kind: Namespace
metadata:
  name: enterprise-app
---
# Database Katmanı (PostgreSQL)
apiVersion: apps/v1
kind: Deployment
metadata:
  name: enterprise-db
  namespace: enterprise-app
  labels:
    app: enterprise-db
spec:
  replicas: 1
  selector:
    matchLabels:
      app: enterprise-db
  template:
    metadata:
      labels:
        app: enterprise-db
    spec:
      containers:
      - name: postgres
        image: postgres:15-alpine
        env:
        - name: POSTGRES_PASSWORD
          value: enterpriseSecret
---
apiVersion: v1
kind: Service
metadata:
  name: db-svc
  namespace: enterprise-app
spec:
  ports:
  - port: 5432
    targetPort: 5432
  selector:
    app: enterprise-db
---
