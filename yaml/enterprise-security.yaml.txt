apiVersion: cilium.io/v2
kind: CiliumNetworkPolicy
metadata:
  name: protect-database-enterprise
  namespace: enterprise-app
spec:
  endpointSelector:
    matchLabels:
      app: enterprise-db
  ingress:
  - toPorts:
    - ports:
      - port: "5432"
        protocol: TCP
    fromEndpoints:
    - matchLabels:
        app: enterprise-backend
---
